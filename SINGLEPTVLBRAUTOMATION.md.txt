LOC_LOADING_OVERLAY = (By.ID, "loading-overlay-id")

# ==========================================
def wait_for_loading_to_finish(driver, timeout=30): 
    try: 
        WebDriverWait(driver, 1.5).until(EC.presence_of_element_located(LOC_LOADING_OVERLAY)) 
        WebDriverWait(driver, timeout).until(EC.invisibility_of_element_located(LOC_LOADING_OVERLAY)) 
    except TimeoutException: 
        pass

def get_visible_elements(driver, xpath, timeout=10): 
    try: 
        WebDriverWait(driver, timeout).until(EC.presence_of_element_located((By.XPATH, xpath))) 
        elements = driver.find_elements(By.XPATH, xpath) 
        return [el for el in elements if el.is_displayed()] 
    except TimeoutException: 
        return []

def type_in_visible_input(driver, xpath, text, timeout=30): 
    end_time = time.time() + timeout 
    while time.time() < end_time: 
        try: 
            visible_elements = get_visible_elements(driver, xpath) 
            if not visible_elements: 
                time.sleep(1) 
                continue
            
            el = visible_elements[0]
            driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", el)
            time.sleep(0.5)
            el.clear()
            el.send_keys(text)
            return 
        except StaleElementReferenceException:
            time.sleep(0.5)
            
    raise Exception(f"Failed to type in visible input for xpath: {xpath}")

def click_visible_element(driver, xpath, timeout=30): 
    end_time = time.time() + timeout 
    while time.time() < end_time: 
        try: 
            visible_elements = get_visible_elements(driver, xpath) 
            if not visible_elements: 
                time.sleep(1) 
                continue
                
            el = visible_elements[0]
            driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", el)
            time.sleep(0.5)
            try:
                el.click()
            except:
                driver.execute_script("arguments[0].click();", el)
            return 
        except StaleElementReferenceException:
            time.sleep(0.5)
            
    raise Exception(f"Failed to click visible element for xpath: {xpath}")

def select_native_dropdown_visible(driver, wait, partial_id, option_text, timeout_sec=20, is_sandbox=False): 
    print(f"    -> Selecting '{option_text}'...") 
    lower_id = partial_id.lower() 
    xpath = f"//select[contains(translate(@id, 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz'), '{lower_id}')]"

    try:
        wait.until(EC.presence_of_element_located((By.XPATH, xpath)))
    except TimeoutException:
        raise Exception(f"Dropdown '{partial_id}' did not appear on page.")
        
    timeout = time.time() + timeout_sec  
    clicked = False

    while time.time() < timeout:
        try:
            wait_for_loading_to_finish(driver)
            visible_elements = [el for el in driver.find_elements(By.XPATH, xpath) if el.is_displayed()]
            
            if not visible_elements:
                time.sleep(1)
                continue
                
            select_element = visible_elements[0]
            select_obj = Select(select_element)
            
            # --- LATEST DATE LOGIC FOR SANDBOXES ---
            if is_sandbox:
                base_match = re.search(r'^(.*?)(?=\d{4}[/-]\d{2}[/-]\d{2})', option_text)
                base_name = base_match.group(1) if base_match else option_text
                
                matching_options = []
                for option in select_obj.options:
                    opt_text = option.text.strip()
                    if opt_text.startswith(base_name):
                        date_match = re.search(r'(\d{4})[/-](\d{2})[/-](\d{2})', opt_text)
                        date_val = int(f"{date_match.group(1)}{date_match.group(2)}{date_match.group(3)}") if date_match else 0
                        matching_options.append((opt_text, date_val, option))
                
                if matching_options:
                    matching_options.sort(key=lambda x: x[1], reverse=True)
                    best_opt_text = matching_options[0][0]
                    
                    driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", select_element)
                    select_obj.select_by_visible_text(best_opt_text)
                    wait_for_loading_to_finish(driver)
                    time.sleep(0.5)
                    
                    if best_opt_text != option_text:
                        print(f"    -> [!] Auto-corrected to latest version on server: '{best_opt_text}'")
                    return best_opt_text

            # --- NORMAL LOGIC FOR EVERYTHING ELSE ---
            for option in select_obj.options:
                opt_str = option.text.strip().lower()
                target_str = option_text.strip().lower()
                
                match_found = False
                
                if opt_str == target_str:
                    match_found = True
                elif re.search(r'\b' + re.escape(target_str) + r'\b', opt_str):
                    match_found = True
                else:
                    clean_opt = re.sub(r'[^a-z0-9]', '', opt_str)
                    clean_tgt = re.sub(r'[^a-z0-9]', '', target_str)
                    if len(clean_tgt) > 5 and clean_tgt in clean_opt:
                        match_found = True
                        
                if match_found:
                    driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", select_element)
                    select_obj.select_by_visible_text(option.text)
                    wait_for_loading_to_finish(driver)
                    time.sleep(0.5)
                    return option.text.strip()
            
            if not clicked:
                driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", select_element)
                try:
                    select_element.click()
                except:
                    driver.execute_script("arguments[0].click();", select_element)
                clicked = True
                wait_for_loading_to_finish(driver)
            
            time.sleep(1)
            
        except StaleElementReferenceException:
            time.sleep(0.5)
            
    raise Exception(f"Option '{option_text}' not found in dropdown '{partial_id}'.")

def run_automation(): 
    os.makedirs(BASE_SAVE_DIR_ROOT, exist_ok=True)

    options = webdriver.EdgeOptions()
    prefs = {
        "download.default_directory": DOWNLOAD_DIR,
        "profile.default_content_setting_values.automatic_downloads": 1,
        "download.prompt_for_download": False

    options.add_experimental_option('excludeSwitches', ['enable-logging'])
    options.add_experimental_option("detach", True)

    driver = webdriver.Edge(options=options)
    driver.maximize_window()

    wait = WebDriverWait(driver, 30)
    sso_wait = WebDriverWait(driver, 120)

            time.sleep(1) 
            driver.execute_script("arguments[0].click();", reports_menu)
            time.sleep(1.5) 
            
            single_ptvl_xpath = "//a[contains(@class, 'ui-menuitem-link') and contains(., 'Single PTVL')]"
            buildability_link = wait.until(EC.presence_of_element_located((By.XPATH, single_ptvl_xpath)))
            driver.execute_script("arguments[0].click();", buildability_link)
                
            wait.until(lambda d: "buildabilityReport.faces" in d.current_url and "Multi" not in d.current_url)
            wait_for_loading_to_finish(driver)
            time.sleep(3)

            successfully_submitted = []

            # ==========================================
            # PHASE 1: BATCH SUBMISSION LOOP
            # ==========================================
            for i, sandbox in enumerate(sandbox_list):
                print(f"\n  ▶ SUBMITTING SANDBOX {i+1}/{len(sandbox_list)}: {sandbox}")

                try:
                    print("  Step 2: Configuring Dropdowns...")
                    
                    dropdown_vehicle = "TRD" if "TRD" in vehicle else vehicle
                    select_native_dropdown_visible(driver, wait, "vehicle", dropdown_vehicle, timeout_sec=15)
                    select_native_dropdown_visible(driver, wait, "year", model_year, timeout_sec=15)
                    
                    actual_sandbox = select_native_dropdown_visible(driver, wait, "sandbox", sandbox, timeout_sec=10, is_sandbox=True)
                    
                    if actual_sandbox in successfully_submitted:
                        print(f"  -> [!] '{actual_sandbox}' was already submitted for {vehicle}. Skipping duplicate.")
                        continue 

                    select_native_dropdown_visible(driver, wait, "view", "Engineering", timeout_sec=15)

                    print("  Step 3: Entering Feature Codes...")
                    type_in_visible_input(driver, "//input[@type='text']", FEATURE_CODES)
                    
                    print("  Step 3: Clicking 'Search by Code' button...")
                    click_visible_element(driver, "//*[normalize-space(text())='Search by Code' or @value='Search by Code']")
                    wait_for_loading_to_finish(driver)

                    print("  Step 3: Clicking 'Select All' (Left Panel)...")
                    click_visible_element(driver, "(//a[normalize-space(text())='All'])[1]")
                    time.sleep(1)

                    print("  Step 3: Clicking 'Add' button...")
                    click_visible_element(driver, "//*[@title='Add' or contains(@src, 'add') or contains(@src, 'Add') or normalize-space(text())='Add' or normalize-space(text())='ADD']")
                    wait_for_loading_to_finish(driver)

                    print("  Step 4: Clicking 'Select All' (Right Panel)...")
                    click_visible_element(driver, "(//a[normalize-space(text())='All'])[2]")
                    time.sleep(1)

                    print("  Step 4: Selecting Display format (Pivot Table)...")
                    select_native_dropdown_visible(driver, wait, "display", "Pivot", timeout_sec=15)

                    print("  Step 4: Clicking 'Submit' button...")
                    click_visible_element(driver, "//*[normalize-space(text())='Submit' or @value='Submit']")
                    wait_for_loading_to_finish(driver)

                    # =========================================================
                    # NEW FIX: Validate Success Banner via Page Text Polling
                    # =========================================================
                    print("  Step 5: Validating success banner...")
                    
                    banner_found = False
                    timeout_end = time.time() + 15
                    
                    while time.time() < timeout_end:
                        try:
                            # Fetch all visible text on the page and make it lowercase
                            body_text = driver.find_element(By.TAG_NAME, "body").text.lower()
                            
                            # Bypass strict DOM structure by just looking for the text directly
                            if "submitted successfully" in body_text:
                                banner_found = True
                                break
                        except StaleElementReferenceException:
                            pass
                        except Exception:
                            pass
                            
                        time.sleep(0.5)

                    if banner_found:
                        print(f"  [✔] Success! Report for {actual_sandbox} successfully queued.")
                    else:
                        error_details = "Success banner 'submitted successfully' did not appear."
                        try:
                            # Try to scrape any red error messages from the screen to log them
                            error_xpath = "//*[contains(@class, 'ui-messages-error') or contains(@class, 'ui-message-error-detail') or contains(@class, 'ui-growl-message')]"
                            err_elements = driver.find_elements(By.XPATH, error_xpath)
                            visible_errs = [el.text.strip() for el in err_elements if el.is_displayed() and el.text.strip()]
                            if visible_errs:
                                error_details = " | ".join(visible_errs)
                        except:
                            pass
                        raise Exception(f"Validation Failed: {error_details}")
                    # =========================================================

                    successfully_submitted.append(actual_sandbox)
                    
                except Exception as e:
                    print(f"  [!] ERROR: {str(e)}")
                    global_skipped_sandboxes.append({
                        "vehicle": vehicle,
                        "sandbox": sandbox,
                        "reason": str(e)
                    })
                    # Refreshing the page clears the error state so the next sandbox can process cleanly
                    driver.refresh()
                    wait_for_loading_to_finish(driver)
                    time.sleep(2)
                    continue

            if not successfully_submitted:
                continue

            # ==========================================
            # PHASE 2: NAVIGATE TO REPORT RETRIEVAL
            # ==========================================
            print("\n  ALL VALID SANDBOXES SUBMITTED. NAVIGATING TO REPORT RETRIEVAL...")
            
            try:
                body = driver.find_element(By.TAG_NAME, "body")
                ActionChains(driver).move_to_element_with_offset(body, 5, 5).perform()
            except:

            reports_menu = wait.until(EC.presence_of_element_located(LOC_REPORTS_MENU))
            ActionChains(driver).move_to_element(reports_menu).perform()
            time.sleep(1)
            driver.execute_script("arguments[0].click();", reports_menu)
            time.sleep(1.5)
            
            retrieval_xpath = "//a[contains(@class, 'ui-menuitem-link') and contains(., 'Report Retrieval')]"
            retrieval_link = wait.until(EC.presence_of_element_located((By.XPATH, retrieval_xpath)))
            driver.execute_script("arguments[0].click();", retrieval_link)
            wait_for_loading_to_finish(driver)
            time.sleep(3)

            # ==========================================
            # PHASE 3: BATCH DOWNLOAD LOOP
            # ==========================================
            num_success = len(successfully_submitted)
            
            for i, actual_sandbox in enumerate(successfully_submitted):
                row_index = num_success - 1 - i
                
                print(f"\n  ▶ DOWNLOADING {vehicle} REPORT FOR: {actual_sandbox} (Targeting Table Row {row_index + 1})")
                
                report_ready = False
                download_link = None
                
                for attempt in range(60): 
                    wait_for_loading_to_finish(driver)
                    
                    try:
                        xpath_text = "//*[contains(translate(normalize-space(text()), 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz'), 'view') or contains(translate(normalize-space(text()), 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz'), 'download')]"
                        all_elements = driver.find_elements(By.XPATH, xpath_text)
                        
                        visible_elements = [el for el in all_elements if el.is_displayed() and el.tag_name.lower() not in ['body', 'html', 'tr', 'td', 'tbody', 'table', 'th', 'thead']]
                        
                        if len(visible_elements) > row_index:
                            target_element = visible_elements[row_index]
                            parent = target_element.find_element(By.XPATH, "..")
                            
                            tag = target_element.tag_name.lower()
                            parent_tag = parent.tag_name.lower()
                            
                            is_active = False
                            element_to_click = target_element
                            
                            if tag in ["a", "button"]:
                                is_active = True
                            elif parent_tag in ["a", "button"]:
                                is_active = True
                                element_to_click = parent
                                
                            if is_active:
                                class_attr = (element_to_click.get_attribute("class") or "").lower()
                                if "disabled" not in class_attr:
                                    download_link = element_to_click
                                    report_ready = True
                                    
                    except StaleElementReferenceException:
                        pass 
                        
                    if report_ready and download_link:
                        break 
                        
                    print(f"    -> 'View/Download' is grey/pending. Clicking on-screen 'Refresh' button... (Attempt {attempt+1}/60)")
                    try:
                        refresh_btns = driver.find_elements(By.XPATH, "//*[normalize-space(text())='Refresh' or @value='Refresh']")
                        visible_refresh = [btn for btn in refresh_btns if btn.is_displayed()]
                        if visible_refresh:
                            driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", visible_refresh[0])
                            time.sleep(0.5)
                            driver.execute_script("arguments[0].click();", visible_refresh[0])
                        else:
                            driver.refresh()
                    except Exception:
                        driver.refresh()
                    
                    wait_for_loading_to_finish(driver)
                    time.sleep(2)
                    
                if not report_ready or not download_link:
                    raise Exception(f"Timed out waiting for the report to generate for {actual_sandbox}.")

                print("    -> Link is blue/clickable! Downloading now...")
                before_downloads = set(os.listdir(DOWNLOAD_DIR))
                
                driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", download_link)
                driver.execute_script("arguments[0].click();", download_link)

                new_file = None
                timeout = time.time() + 60

                    current_downloads = set(os.listdir(DOWNLOAD_DIR))
                    new_files = current_downloads - before_downloads
                    if new_files:
                        potential_file = list(new_files)[0]
                        if not potential_file.endswith('.crdownload') and not potential_file.endswith('.tmp'):
                            new_file = potential_file
                            break
                    time.sleep(1)

                if not new_file:
                    raise Exception("File download timed out or failed.")

                safe_sandbox_name = actual_sandbox.replace('/', '-')
                safe_sandbox_name = re.sub(r'[\\/*?:"<>|]', '_', safe_sandbox_name)

                crf_match = re.search(r'\d{8}', actual_sandbox)
                if crf_match:
                    crf_number = crf_match.group(0)
                    target_folder = os.path.join(BASE_SAVE_DIR_ROOT, vehicle, f"CRF-{crf_number}")
                else:
                    target_folder = os.path.join(BASE_SAVE_DIR_ROOT, vehicle, safe_sandbox_name)
                    
                os.makedirs(target_folder, exist_ok=True)
                
                source_path = os.path.join(DOWNLOAD_DIR, new_file)
                _, ext = os.path.splitext(new_file)
                final_file_name = f"{safe_sandbox_name}{ext}"
                final_path = os.path.join(target_folder, final_file_name)
                
                if os.path.exists(final_path):
                    os.remove(final_path)
                    
                shutil.move(source_path, final_path)
                print(f"    -> SUCCESS! File saved to: {final_path}")

        print("\n=======================================================")
        print("ALL VEHICLES AND REPORTS PROCESSED!")
        print("=======================================================")

        if global_skipped_sandboxes:
            print("\n=======================================================")
            print("⚠️ SKIPPED SANDBOXES SUMMARY")
            print("=======================================================")
            for skipped in global_skipped_sandboxes:
                print(f" - Vehicle: {skipped['vehicle']} | Sandbox: {skipped['sandbox']}")
                print(f"   Reason:  {skipped['reason']}")
            print("=======================================================\n")
        else:
            print("\n✅ All sandboxes were processed successfully. No errors!\n")

    except Exception as e:
        print(f"\n[!] CRITICAL ERROR: {str(e)}")
        traceback.print_exc()
        error_dir = os.path.join(BASE_SAVE_DIR_ROOT, "results")
        os.makedirs(error_dir, exist_ok=True)
        screenshot_path = os.path.join(error_dir, "buildability_error_screenshot.png")
        try:
            driver.save_screenshot(screenshot_path)
            print(f"[!] A screenshot of the failure has been saved to: {screenshot_path}\n")
        except:
            pass
            
    finally:
        print("Script execution finished. Browser will remain open.")

if __name__ == '__main__':
    run_automation()
