def wait_for_loading_to_finish(driver, timeout=30): 
    try: 
        WebDriverWait(driver, 1.5).until(EC.presence_of_element_located((By.ID, "loading-overlay-id"))) 
        WebDriverWait(driver, timeout).until(EC.invisibility_of_element_located((By.ID, "loading-overlay-id"))) 
    except TimeoutException: 
        pass

def get_visible_elements(driver, xpath, timeout=10): 
    try: 
        WebDriverWait(driver, timeout).until(EC.presence_of_element_located((By.XPATH, xpath))) 
        elements = driver.find_elements(By.XPATH, xpath) 
        return [el for el in elements if el.is_displayed()] 
    except TimeoutException: 
        return []

def click_visible_element(driver, xpath, timeout=30): 
    end_time = time.time() + timeout 
    while time.time() < end_time: 
        try: 
            visible_elements = get_visible_elements(driver, xpath) 
            if not visible_elements: 
                time.sleep(1) 
                continue
            el = visible_elements[0]
            driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", el)
            time.sleep(0.5)
            try:
                el.click()
            except:
                driver.execute_script("arguments[0].click();", el)
            return 
        except StaleElementReferenceException:
            time.sleep(0.5)
    raise Exception(f"Failed to click visible element for xpath: {xpath}")

def type_in_visible_input(driver, xpath, text, timeout=30): 
    end_time = time.time() + timeout 
    while time.time() < end_time: 
        try: 
            visible_elements = get_visible_elements(driver, xpath) 
            if not visible_elements: 
                time.sleep(1) 
                continue
            
            el = visible_elements[0]
            
            if el.tag_name.lower() not in ['input', 'textarea']:
                inputs = el.find_elements(By.XPATH, ".//input | .//textarea")
                if inputs:
                    el = inputs[0]
            
            driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", el)
            time.sleep(0.5)
            
            try:
                el.clear()
            except:
                driver.execute_script("arguments[0].value = '';", el)
                
            try:
                el.send_keys(text)
            except:
                driver.execute_script("arguments[0].value = arguments[1];", el, text)
                
            return 
        except StaleElementReferenceException:
            time.sleep(0.5)
    raise Exception(f"Failed to type in visible input for xpath: {xpath}")

def run_automation():

    options.add_argument("--log-level=3")
    options.add_experimental_option('excludeSwitches', ['enable-logging'])
    options.add_experimental_option("detach", True)

    driver = webdriver.Edge(options=options)
    driver.maximize_window()

        driver.get(BASE_URL)

        print(" -> Page body loaded. Waiting 3 seconds for scripts...")
        time.sleep(3) 

        print("Step 1.1: Clicking Reports dropdown menu...")
        wait_for_loading_to_finish(driver)
        
        try:
            reports_menu = wait.until(EC.presence_of_element_located((By.XPATH, XPATHS["REPORTS_DROPDOWN_MENU"])))
            driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", reports_menu)
            time.sleep(1)
            driver.execute_script("arguments[0].click();", reports_menu)
            print(" -> Click executed. Waiting 2 seconds for dropdown animation...")
            time.sleep(2)
        except Exception as e:
            print(f" -> Warning on 1.1: {e}")

        print("Step 1.2: Clicking Buildability Report - Multi PTVL...")
        try:
            multi_link = wait.until(EC.presence_of_element_located((By.XPATH, XPATHS["MULTI_PTVL_REPORT_LINK"])))
            driver.execute_script("arguments[0].click();", multi_link)
        except TimeoutException:
            print(" -> Submenu item not found! Retrying Reports menu click...")
            driver.execute_script("arguments[0].click();", reports_menu)
            time.sleep(2)
            multi_link = wait.until(EC.presence_of_element_located((By.XPATH, XPATHS["MULTI_PTVL_REPORT_LINK"])))
            driver.execute_script("arguments[0].click();", multi_link)
            
        print(" -> Submenu clicked. Waiting for next page to load...")

        # --- LOGIC 1: Loop through Standard PTVLs ---

            print("  Step 4: Clicking Search button...")
            click_visible_element(driver, XPATHS["PTVL_SEARCH_BTN"])
            wait_for_loading_to_finish(driver)

            print(f"  Step 5: Selecting checkbox for MY {current_my} in Applicable PTVL/MY table...")

                driver.execute_script("arguments[0].click();", options_list[-1])

        click_visible_element(driver, XPATHS["SUBMIT_BTN"])
        wait_for_loading_to_finish(driver)

        # =========================================================
        # Validate Success Banner

        print("Step 14.5: Validating success banner...")
        banner_found = False
        timeout_end = time.time() + 60  
        
        while time.time() < timeout_end:
            try:
                body_text = driver.find_element(By.TAG_NAME, "body").text.lower()

                pass

        print("Step 15: Navigating to Report Retrieval...")

        driver.execute_script("arguments[0].click();", retrieval_link)
        wait_for_loading_to_finish(driver)
        time.sleep(3)

        print("Step 16: Waiting for report to generate and downloading...")
        report_ready = False
        download_link = None
        
        for attempt in range(60): 
            wait_for_loading_to_finish(driver)
            try:
                xpath_text = "//*[contains(translate(normalize-space(text()), 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz'), 'view') or contains(translate(normalize-space(text()), 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz'), 'download')]"
                visible_elements = [el for el in driver.find_elements(By.XPATH, xpath_text) if el.is_displayed() and el.tag_name.lower() not in ['body', 'html', 'tr', 'td', 'tbody', 'table', 'th', 'thead']]
                
                if visible_elements:
                    target_element = visible_elements[0] 
                    parent = target_element.find_element(By.XPATH, "..")
                    
                    tag = target_element.tag_name.lower()
                    parent_tag = parent.tag_name.lower()
                    
                    is_active = False
                    element_to_click = target_element
                    
                    if tag in ["a", "button"]: is_active = True
                    elif parent_tag in ["a", "button"]:
                        is_active = True
                        element_to_click = parent
                        
                    if is_active and "disabled" not in (element_to_click.get_attribute("class") or "").lower():
                        download_link = element_to_click
                        report_ready = True
                            
            except StaleElementReferenceException:
                pass 
                
            if report_ready and download_link:
                break 
                
            print(f" -> Refreshing view... (Attempt {attempt+1}/60)")
            try:
                refresh_btns = driver.find_elements(By.XPATH, "//*[normalize-space(text())='Refresh' or @value='Refresh']")
                visible_refresh = [btn for btn in refresh_btns if btn.is_displayed()]
                if visible_refresh:
                    driver.execute_script("arguments[0].click();", visible_refresh[0])
                else:
                    driver.refresh()
            except Exception:
                driver.refresh()
            
            wait_for_loading_to_finish(driver)
            time.sleep(2)
            
        if report_ready and download_link:
            print(" -> Link is active. Downloading...")
            
            before_downloads = set(os.listdir(DOWNLOAD_DIR))
            
            driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", download_link)
            driver.execute_script("arguments[0].click();", download_link)
            
            print(" -> Waiting for file download to complete...")
            new_file = None
            timeout = time.time() + 60 
            while time.time() < timeout:
                current_downloads = set(os.listdir(DOWNLOAD_DIR))
                new_files = current_downloads - before_downloads
                if new_files:
                    potential_file = list(new_files)[0]
                    if not potential_file.endswith('.crdownload') and not potential_file.endswith('.tmp'):
                        new_file = potential_file
                        break
                time.sleep(1)

            if new_file:
                source_path = os.path.join(DOWNLOAD_DIR, new_file)
                
                # --- Format new file name: multiPTVLBR_DD-MM-YY ---
                file_extension = os.path.splitext(new_file)[1]
                date_str = datetime.now().strftime("%d-%m-%y")
                new_filename = f"multiPTVLBR_{date_str}{file_extension}"
                
                final_path = os.path.join(BASE_SAVE_DIR_ROOT, new_filename)
                
                if os.path.exists(final_path):
                    os.remove(final_path)
                shutil.move(source_path, final_path)
                print(f"Download complete! File saved to: {final_path}")
            else:
                print("Download timed out.")
        else:
            print("Report generation timed out.")

    except Exception as e:
        print(f"[!] ERROR: {str(e)}")
        traceback.print_exc()
    finally:
        print("Script execution finished. Browser will remain open.")

if __name__ == '__main__':
    run_automation()
